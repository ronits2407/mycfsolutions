import java.util.Scanner
 
fun solve_ronit(scanner: Scanner) {
    val n = scanner.nextInt()
 
    if (n % 5 == 2 || n % 5 == 4) {
        println("NO")
        return
    } else {
        println("YES")
    }
 
    val ans = mutableListOf<Pair<Char, Char>>()
    
    when (n % 5) {
        0 -> {
            val count = n / 5
            for (i in 0 until count) {
                ans.add(Pair('.', '.'))
                ans.add(Pair('.', '*'))
                ans.add(Pair('.', '.'))
            }
        }
        1 -> {
            val count = (n - 1) / 5
            ans.add(Pair('.', '*'))
            for (i in 0 until count) {
                ans.add(Pair('.', '.'))
                ans.add(Pair('.', '.'))
                ans.add(Pair('.', '*'))
            }
        }
        3 -> {
            val count = (n - 3) / 5
            ans.add(Pair('.', '.'))
            ans.add(Pair('.', '*'))
            for (i in 0 until count) {
                ans.add(Pair('.', '.'))
                ans.add(Pair('.', '.'))
                ans.add(Pair('.', '*'))
            }
        }
    }
 
    println(ans.size)
 
    // Using StringBuilder for fast I/O to avoid Time Limit Exceeded (TLE)
    val row1 = StringBuilder()
    val row2 = StringBuilder()
    
    for (i in 0 until ans.size) {
        row1.append(ans[i].first)
        row2.append(ans[i].second)
    }
    
    println(row1.toString())
    println(row2.toString())
}
 
fun main() {
    val scanner = Scanner(System.`in`)
    if (scanner.hasNextInt()) {
        val t = scanner.nextInt()
        repeat(t) {
            solve_ronit(scanner)
        }
    }
}
